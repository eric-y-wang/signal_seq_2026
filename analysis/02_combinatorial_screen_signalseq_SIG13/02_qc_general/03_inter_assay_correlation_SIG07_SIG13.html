Based on this, threshold of 100 significant interactions per group in both SIG07 and SIG13 seems reasonable

interactions_to_keep = nsig_interactions %>% filter(nsig.sig13 >= 100 & nsig.sig07 >= 100) %>% pull(interaction_sorted)

Take mean of duplicate interaction terms

sig13_to_corr <- sig13_data_interact %>%
  group_by(interaction_sorted, name) %>%
  summarise(mean_interaction_lfc = mean(lfc_interaction, na.rm = TRUE))
## `summarise()` has grouped output by 'interaction_sorted'. You can override
## using the `.groups` argument.
sig07_to_corr <- sig07_data_interact %>%
  group_by(interaction_sorted, name) %>%
  summarise(mean_interaction_lfc = mean(lfc_interaction, na.rm = TRUE))
## `summarise()` has grouped output by 'interaction_sorted'. You can override
## using the `.groups` argument.
sig13_to_corr <- as_data_frame(sig13_to_corr)
sig07_to_corr <- as_data_frame(sig07_to_corr)
joint_data_interact <- inner_join(sig13_to_corr, sig07_to_corr, by = c("interaction_sorted", "name"), suffix = c(".sig13", ".sig07")) 
joint_data_interact_filtered <- joint_data_interact %>% filter(interaction_sorted %in% interactions_to_keep)

Get correlation

correlation_results <- joint_data_interact_filtered %>%
  group_by(interaction_sorted) %>%
  summarise(
    pearson_corr = cor(mean_interaction_lfc.sig13, mean_interaction_lfc.sig07, method = "pearson")
  )

Get correlation grouped by interaction and ligand

ligand_correlations_singles <- joint_singles %>%
  group_by(interaction) %>% 
  summarise(
    pearson_corr = cor(lfc.sig13, lfc.sig07, method = "pearson"),
    # It is often helpful to count how many genes are driving this correlation
    n_genes = n(), 
    .groups = "drop"
  ) %>%
  arrange(desc(pearson_corr))
