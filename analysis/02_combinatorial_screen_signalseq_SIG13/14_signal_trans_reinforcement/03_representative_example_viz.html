IL6/21 + CCL19/21A

These combinations are interesting because IL6 and IL21 both signal predominatly through STAT3. However, IL21 has strong synergistic effects in combination with CCR7 ligands (CCL19/21A) while IL6 has weak synergistic effects. Here, we examine whether signal reinforcement is differentially present between the two types of combinations.

condition_sub <- c("IL21_linker","IL6_linker","linker_CCL19","linker_CCL21A")
receptors <- c("Il21r","Il6ra","Ccr7")

deg_sub <- deg_condition %>%
  filter(condition %in% condition_sub) %>%
  filter(name %in% receptors) %>%
  mutate(condition = gsub("_linker|linker_","",condition))

res_sub <- res %>%
  filter(interaction %in% c("IL21_CCL19","IL21_CCL21A","IL6_CCL19","IL6_CCL21A"))
sig_stars <- function(p) {
  case_when(
    p < 0.0001 ~ "****",
    p < 0.001 ~ "***",
    p < 0.01  ~ "**",
    p < 0.05  ~ "*",
    TRUE      ~ ""
  )
}

p1 <- deg_sub %>%
  filter(name == "Ccr7") %>%
  filter(condition %in% c("IL21","IL6")) %>%
  mutate(stars = sig_stars(adj_pval)) %>%
  ggplot(aes(x = condition, y = name)) +
  geom_point(aes(fill = lfc, size = -log10(adj_pval)), shape = 21) +
  geom_text(aes(label = stars), nudge_y = 0.2, size = 4, vjust = 0) +
  scale_fill_distiller(palette = "RdBu", limits = c(-0.45, 0.45)) +
  scale_size_continuous(limits = c(0, 16)) +
  coord_equal() +
  theme(axis.title.x = element_blank(),
        axis.title.y = element_blank())

p2 <- deg_sub %>%
  filter(name %in% c("Il21r","Il6ra")) %>%
  filter(condition %in% c("CCL19","CCL21A")) %>%
  mutate(stars = sig_stars(adj_pval)) %>%
  ggplot(aes(x = condition, y = name)) +
  geom_point(aes(fill = lfc, size = -log10(adj_pval)), shape = 21) +
  geom_text(aes(label = stars), nudge_y = 0.2, size = 4, vjust = 0) +
  scale_fill_distiller(palette = "RdBu", limits = c(-0.45, 0.45)) +
  scale_size_continuous(limits = c(0, 16)) +
  coord_equal() +
  theme(axis.title.x = element_blank(),
        axis.title.y = element_blank())

p1 + p2 + plot_layout(ncol = 1, guides = "collect") + plot_annotation("IL6/21 + CCL19/21A receptor cross-regulation")
